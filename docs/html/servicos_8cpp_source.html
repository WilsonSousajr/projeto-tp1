12// Converte abreviação de mês (ex.: JAN) para número (1-12)
13int monthFromAbbr(const string &m) {
14 static const map<string, int> mm{
15 // Português
16 {"JAN", 1}, {"FEV", 2}, {"MAR", 3}, {"ABR", 4}, {"MAI", 5}, {"JUN", 6},
17 {"JUL", 7}, {"AGO", 8}, {"SET", 9}, {"OUT", 10}, {"NOV", 11}, {"DEZ", 12},
18 // Inglês (aceito também)
19 {"FEB", 2}, {"APR", 4}, {"MAY", 5}, {"AUG", 8}, {"SEP", 9}, {"OCT", 10}, {"DEC", 12}
20 };
21 string u = m;
22 for (char &c : u) {
23 c = static_cast<char>(toupper(static_cast<unsigned char>(c)));
24 }
25 auto it = mm.find(u);
26 if (it == mm.end()) {
27 throw invalid_argument("Mês inválido.");
28 }
29 return it->second;
30}

32// Faz parsing de data no formato D/DD-MMM-YYYY
33bool parseDateDMY(const string &s, int &d, int &m, int &y) {
34 size_t p1 = s.find('-');
35 size_t p2 = s.find('-', (p1 == string::npos) ? 0 : p1 + 1);
36 if (p1 == string::npos || p2 == string::npos) {
37 return false;
38 }
39 d = stoi(s.substr(0, p1));
40 m = monthFromAbbr(s.substr(p1 + 1, p2 - p1 - 1));
41 y = stoi(s.substr(p2 + 1));
42 return true;
43}

45// Número de dias serial para comparação de intervalos
46long dayNumber(int d, int m, int y) {
47 if (m < 3) {
48 y--;
49 m += 12;
50 }
51 return 365L * y + y / 4 - y / 100 + y / 400 + (153 * (m - 3) + 2) / 5 + d - 1;
52}

54struct Range {
55 long a;
56 long b;
57};
58
59Range makeRange(const string &start, int dias) {
60 int d, m, y;
61 if (!parseDateDMY(start, d, m, y)) {
62 throw invalid_argument("Data inválida.");
63 }
64 long s = dayNumber(d, m, y);
65 long e = s + ((dias <= 0) ? 0 : (dias - 1));
66 return {s, e};
67}

69bool overlaps(const Range &r1, const Range &r2) {
70 return !(r1.b < r2.a || r2.b < r1.a);
